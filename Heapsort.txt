#include <iostream> 
using namespace std; 
class Heap { 
public: 
int arr[100]; 
void getArray(int n); 
void printArray(int n); 
void maxHeapify(int arr[], int n, int i); 
void maxHeapSort(int arr[], int n); 
void minHeapify(int arr[], int n, int i); 
void minHeapSort(int arr[], int n); 
}; 
// Input array 
void Heap::getArray(int n) { 
for (int i = 0; i < n; i++) { 
cin >> arr[i]; 
} 
} 
// Print array 
void Heap::printArray(int n) { 
for (int i = 0; i < n; i++) { 
cout << arr[i] << " "; 
} 
cout << endl; 
} 
// Max Heapify 
void Heap::maxHeapify(int arr[], int n, int i) { 
int max = i; 
int left = 2 * i + 1; 
int right = 2 * i + 2; 
if (left < n && arr[left] > arr[max]) 
max = left; 
if (right < n && arr[right] > arr[max]) 
max = right; 
if (max != i) { 
swap(arr[i], arr[max]); 
maxHeapify(arr, n, max); 
} 
} 
// Max Heap Sort 
void Heap::maxHeapSort(int arr[], int n) { 
for (int i = n / 2 - 1; i >= 0; i--) { 
maxHeapify(arr, n, i); 
} 
for (int i = n - 1; i >= 0; i--) { 
swap(arr[0], arr[i]); 
maxHeapify(arr, i, 0); 
} 
} 
// Min Heapify 
void Heap::minHeapify(int arr[], int n, int i) { 
int min = i; 
int left = 2 * i + 1; 
int right = 2 * i + 2; 
if (left < n && arr[left] < arr[min]) 
min = left; 
if (right < n && arr[right] < arr[min]) 
min = right; 
if (min != i) { 
swap(arr[i], arr[min]); 
minHeapify(arr, n, min); 
} 
} 
// Min Heap Sort 
void Heap::minHeapSort(int arr[], int n) { 
for (int i = n / 2 - 1; i >= 0; i--) { 
minHeapify(arr, n, i); 
} 
for (int i = n - 1; i >= 0; i--) { 
swap(arr[0], arr[i]); 
minHeapify(arr, i, 0); 
} 
} 
// Main function 
int main() { 
Heap h; 
int n, ch; 
cout << "Heap Sort Program" << endl; 
do { 
cout << "\nEnter the Number of Elements: "; 
cin >> n; 
cout << "Enter the Array Elements:\n"; 
h.getArray(n); 
cout << "\nArray Before Sorting: "; 
h.printArray(n); 
h.maxHeapSort(h.arr, n); 
cout << "Array After Sorting (Max Heap): "; 
h.printArray(n); 
h.minHeapSort(h.arr, n); 
cout << "Array After Sorting (Min Heap): "; 
h.printArray(n); 
cout << "\nDo You Want to Continue? Press 1 Else 0: "; 
cin >> ch; 
} while (ch == 1); 
return 0; 
}
